\begin{abstract}
Finding the slope of the tangent line to $y=x^2$ at a point, and at an arbitrary point. Textbook §2.1.
\end{abstract}
\maketitle

\begin{exercise}\label{exercise:2-9}
Suppose we have the function $f(x)=x^{2}$. How do we find the slope of the tangent line at $x=1$?
\begin{image}
\includegraphics[width=0.5\textwidth]{b48a5324-93dc-45ee-b7c2-dc0b5706bda6-016_567_737_606_692}
\end{image}
% open work area in the original; result is determinable
The slope of the tangent line at $x=1$ is $\answer{2}$.

Now try this one: find the slope of the tangent line at
% open work area in the original; results are determinable
\begin{itemize}
  \item $x=0$: slope $=\answer{0}$
  \item $x=2$: slope $=\answer{4}$
  \item $x=-1$: slope $=\answer{-2}$
\end{itemize}

Can you come up with a formula for the slope of the tangent line for arbitrary $x$?
% open work area in the original; result is determinable
\[
\text{slope}=\answer{2x}
\]
\end{exercise}
