% Long s (U+017F), used once in the remark on the origin of the integral sign.
%
% The Ximera build runs LuaTeX for BOTH pdf and html, and U+017F is in LuaTeX's
% default Unicode font, so the character passes straight through and nothing
% below is needed there.  Only an 8-bit engine (a local `pdflatex foo.tex`) has
% to have it mapped -- and only that path wants yfonts, which is absent from the
% Ximera Docker image's partial TeX Live.  Loading it unconditionally fails every
% document in CI, so both the package and the mapping are guarded.
\IfFileExists{yfonts.sty}{%
  \usepackage{yfonts}%
  \newcommand{\longs}{\textfrak{s}}%  Fraktur lowercase s is the long-s glyph
}{%
  \newcommand{\longs}{s}%              no Fraktur font available; fall back to a plain s
}
\ifdefined\directlua\else
  \usepackage{newunicodechar}
  \newunicodechar{ſ}{\longs}
\fi
